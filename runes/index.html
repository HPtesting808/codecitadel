🌀 Warp Workflow

# ~/.warp/workflows/pr-review.yaml
name: "PR Review"
description: "Create PR with AI review"
command: |
  gh pr create --fill --web
  && gh pr view --json number,headRefName -q '.number + " " + .headRefName'
  | xargs -I {} sh -c 'aider --message "Review PR #{}"'
tags: ["git", "ai", "review"]

📜 Git Aliases

# ~/.gitconfig
[alias]
  co = checkout
  cb = checkout -b
  st = status -sb
  lg = log --oneline --graph --decorate -20
  lga = log --oneline --graph --decorate --all
  amend = commit --amend --no-edit
  undo = reset --soft HEAD~1
  wip = commit -am "WIP"
  unwip = reset HEAD~1
  branches = branch -v
  cleanup = !git branch --merged | grep -v \"\\*\" | xargs -n 1 git branch -d

🖥️ tmux Dev Layout

# ~/.tmux/dev-layout.sh
#!/bin/bash
session="dev"
tmux new-session -d -s $session -n editor
tmux send-keys -t $session:editor 'nvim' Enter
tmux split-window -h -t $session:editor -p 30
tmux send-keys -t $session:editor.1 'aider' Enter
tmux split-window -v -t $session:editor.0 -p 25
tmux send-keys -t $session:editor.0 'git status' Enter
tmux attach -t $session
